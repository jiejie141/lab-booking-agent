# ===========================================================================
# lab-booking-agent 运行依赖
# ===========================================================================
# 说明：Web 控制台（src/lagent/web/index.html）是零依赖单文件，
#      不占任何 pip 包；下面的依赖只服务后端与测试。
#
# ⚠ 这里有【两套】文件，用途不同，别混用：
#
#   requirements.txt   ← 你正在看的这份。写的是**直接依赖 + 兼容区间**。
#                        给人读的：一眼看清这个项目用了哪些库。
#                        装它 = 拿到一个大致能跑的环境。
#
#   requirements.lock  ← 机器生成的**全量精确锁定**（含传递依赖）。
#                        给 CI / 部署用：版本完全定死，装出来一模一样。
#                        生成方式见那个文件头的注释。
#
# 为什么要分成两套 —— 起因是一次真实排查：
#   CI 的 py3.13 job 报 mypy 错、py3.10 job 全绿，两边跑的是同一份 mypy。
#   真因是本文件当时只写了 >= 下限：SQLAlchemy 2.1 抬高了 requires_python，
#   于是 py3.10 被解析到 2.0、py3.13 拿到 2.1 —— **两个 job 静默地在测两套栈**。
#   当时的处置是在 CI 里加一步打印版本，但那只让问题**可见**，没让它**不可发生**。
#
#   现在补上两件事：
#     1) 直接依赖加**上限**（下一个大版本可能破坏兼容，见下方各条注释）；
#     2) 全量锁定到 requirements.lock，CI 与部署装那份。
# ---------------------------------------------------------------------------

# --- 服务层 ---
# 上限取当前实测通过的下一个主版本。FastAPI 到 1.x 会再有一次破坏性调整，
# 到那天必须**有意识地**升级并重跑全部测试，而不是被 pip 悄悄带上去。
fastapi>=0.115,<1.0
uvicorn>=0.30,<1.0

# --- Agent 编排 ---
# langgraph 目前是 1.x（0.x → 1.x 中间改过 API）。锁在主版本内最稳。
langgraph>=0.2,<2.0

# --- 数据层（SQLite 本地默认 / PostgreSQL 走 docker-compose） ---
# ★ SQLAlchemy 的 2.0 → 2.1 正是上面那次事故的直接原因。
#   2.1 抬高了 requires_python，导致 py3.10 装不上 —— 版本漂移从此开始。
#   这里锁到 <2.2：要上 2.1 就显式改这一行，改完必须在两个 Python 版本上各跑一遍。
sqlalchemy[asyncio]>=2.0,<2.2
aiosqlite>=0.20,<1.0
# 迁移：结构变更唯一被允许的入口（见 migrations/README.md）
alembic>=1.13,<2.0

# --- 配置与模型 ---
pydantic>=2.7,<3.0
pydantic-settings>=2.3,<3.0

# --- 时区数据 ---
# Windows 不自带系统时区库，zoneinfo 需要它才能解析 Asia/Shanghai。
# 缺失时 clock.py 会退回固定 +8 偏移（国内无夏令时，等价），不会崩。
# tzdata 是无版本语义的数据包（IANA 数据库），每年更新几次，不按 semver 走 ——
# 所以上限写成 <3000 而不是 <2027：这里要的是"绝对不可能撞上的哨兵"，
# 而不是假装预知下一个版本号。它只用来让 CI 的边界校验有一侧可比。
tzdata>=2024.1,<3000

# --- 检索与 HTTP ---
# BM25 为手写实现，不需要 rank_bm25；
# httpx 用于「真实模型」模式（直连 OpenAI 兼容接口，因此不需要 langchain-openai）
httpx>=0.27,<1.0
pyyaml>=6.0,<7.0

# --- 测试 ---
# pytest 8 → 9 有一次较大的内部调整；本项目实测在 9.1 下全绿，
# 但区间仍留 8，方便别人的机器先装 8 也能跑。
pytest>=8.0,<10.0
pytest-asyncio>=0.24,<2.0
